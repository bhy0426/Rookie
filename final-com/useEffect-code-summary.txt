Home.tsx 62번째 useEffect 코드 요약

useEffect(() => {
  const handleWheel = (event) => {
    // 1. 현재 섹션과 휠 방향 확인
    const current = currentSectionRef.current;
    const isScrollDown = event.deltaY > 6;
    const isScrollUp = event.deltaY < -6;

    if (!isScrollDown && !isScrollUp) return;

    // 2. 첫 섹션 / 마지막 섹션 여부 확인
    const isLastSection = current === totalSections - 1;
    const isFirstSection = current === 0;

    // 3. 일반 스크롤 모드면 휠 제어하지 않음
    if (scrollModeRef.current === "free") return;

    // 4. 잠금 복귀 대기 중이면 기본 스크롤만 차단
    if (scrollModeRef.current === "relockPending") {
      event.preventDefault();
      return;
    }

    // 5. 마지막 섹션에서 아래로 스크롤하면 일반 스크롤 모드로 전환
    if (isScrollDown && isLastSection) {
      scrollModeRef.current = "free";
      setHomeNavHidden(true);
      return;
    }

    // 6. 홈 섹션 이동 중에는 브라우저 기본 스크롤 차단
    event.preventDefault();

    // 7. 휠 방향에 따라 홈 네비게이션 표시 여부 변경
    if (isScrollUp) setHomeNavHidden(false);
    if (isScrollDown) setHomeNavHidden(true);

    // 8. 연속 휠 입력으로 여러 섹션이 넘어가는 것 방지
    if (wheelLockRef.current) return;
    if (isScrollUp && isFirstSection) return;

    wheelLockRef.current = true;

    // 9. 다음 섹션 계산 후 상태 업데이트
    const nextSection = isScrollDown ? current + 1 : current - 1;

    currentSectionRef.current = nextSection;
    setCurrentSection(nextSection);

    // 10. 일정 시간 후 휠 잠금 해제
    setTimeout(() => {
      wheelLockRef.current = false;
    }, 980);
  };

  // 11. wheel 이벤트 등록
  window.addEventListener("wheel", handleWheel, { passive: false });

  // 12. 컴포넌트가 사라질 때 wheel 이벤트 제거
  return () => {
    window.removeEventListener("wheel", handleWheel);
  };
}, [totalSections, setHomeNavHidden]);


PPT용 한 줄 요약

마우스 휠 방향을 감지해 현재 섹션 번호를 계산하고,
기본 스크롤을 막은 뒤 currentSection을 변경하여
홈 화면을 한 번에 한 섹션씩 이동시키는 useEffect입니다.
